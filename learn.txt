void showPlaylist(char playlist[][100], int playlistSize) {
    printf("\n\033[0;36mPlaylist:\033[0m\n");
    for (int i = 0; i < playlistSize; i++) {
        printf("  %d. %s\n", i + 1, playlist[i]);
    }
}
void stopCurrentSong() {
    if (strlen(currentSongAlias) > 0) {
        char command[100];
        sprintf(command, "close %s", currentSongAlias);
        mciSendString(command, NULL, 0, NULL);
        currentSongAlias[0] = '\0';
    }
}
void playSong(const char* song) {
    stopCurrentSong();
    char command[200];
    sprintf(currentSongAlias, "song%ld", time(NULL));
    sprintf(command, "open \"%s\" type mpegvideo alias %s", song, currentSongAlias);
    mciSendString(command, NULL, 0, NULL);
    sprintf(command, "play %s", currentSongAlias);
    mciSendString(command, NULL, 0, NULL);
    printf("\nNow playing: %s\n", song);
}
void interactivePlayback(char playlist[][100], int size, int shuffle) {
    int indices[MAX_SONGS];
    for (int i = 0; i < size; i++) indices[i] = i;
    if (shuffle) {
        for (int i = 0; i < size; i++) {
            int r = rand() % size;
            int temp = indices[i];
            indices[i] = indices[r];
            indices[r] = temp;
        }
    }
    int current = 0;
    while (current < size && current >= 0) {
        playSong(playlist[indices[current]]);
        printf("\033[1;31mType 'n' for next, 'p' for previous, or 's' to stop and return to menu:\033[0m\n");
        char input[10];
        fgets(input, sizeof(input), stdin); // Clear buffer
        fgets(input, sizeof(input), stdin);
        if (input[0] == 'n' || input[0] == 'N') {
            current = (current + 1) % size;
        } else if (input[0] == 'p' || input[0] == 'P') {
            current = (current - 1 + size) % size;
        } else if (input[0] == 's' || input[0] == 'S') {
            stopCurrentSong();
            printf("Stopped. Returning to menu...\n");
            Sleep(1000);
            return;
        } else {
            printf("Invalid input, try again.\n");
        }
    }
    stopCurrentSong();
    Sleep(1000);
}
void createUserPlaylist(char playlist[][100], int playlistSize, char userPlaylist[][100], int *userPlaylistSize) {
    *userPlaylistSize = 0;
    int choice;
    while (1) {
        system("cls");
        printf("\n\033[0;36mMain Playlist:\033[0m\n");
        for (int i = 0; i < playlistSize; i++) {
            printf("  %d. %s\n", i + 1, playlist[i]);
        }
        printf("\n\033[1;32mEnter song number to add to your playlist (0 to finish): \033[0m");
        scanf("%d", &choice);
        if (choice == 0) break;
        if (choice < 1 || choice > playlistSize) {
            printf("Invalid song number.\n");
            Sleep(1000);
        } else {
            strcpy(userPlaylist[*userPlaylistSize], playlist[choice - 1]);
            (*userPlaylistSize)++;
            printf("Added: %s\n", playlist[choice - 1]);
            Sleep(500);
        }
    }
    printf("\n\033[1;32mYour playlist has been created with %d song(s)!\033[0m\n", *userPlaylistSize);
    system("pause");
}

void searchAndPlaySong(char playlist[][100], int playlistSize) {
    char search[100];
    int foundIndex = -1;
    printf("\n\033[1;34mEnter part of the song name to search: \033[0m");
    scanf("%s", search);
    for (int i = 0; i < playlistSize; i++) {
        if (strstr(playlist[i], search) != NULL) {
            printf("\033[0;32mFound:\033[0m %s\n", playlist[i]);
            foundIndex = i;
            break;
        }
    }
    if (foundIndex != -1) {
        playSong(playlist[foundIndex]);
        printf("\033[1;31mPress ENTER to finish, or type 's' and press ENTER to stop song...\033[0m\n");
        char input[10];
        fgets(input, sizeof(input), stdin); // flush buffer
        fgets(input, sizeof(input), stdin);
        if (input[0] == 's' || input[0] == 'S') {
            stopCurrentSong();
            printf("Stopped. Returning to menu...\n");
            Sleep(1000);
        }
    } else {
        printf("No song found matching '%s'.\n", search);
        Sleep(1000);
    }
}

        "NUMB.mp3", "Criminal.mp3", "Fever.mp3", "IDGAF.mp3", "Jhol.mp3", "Maniac.mp3",

        "Havana.mp3", "Sucker.mp3"
    };

    int userPlaylistSize = 0;
    int playlistSize = 30;
    int choice;
    srand(time(NULL));
    while (1) {
        printMenu();
        printf("\n\033[0;33mEnter your choice: \033[0m");
        scanf("%d", &choice);
        switch (choice) {
            case 1: // Play Playlist in Queue (main playlist, not shuffled)
                interactivePlayback(playlist, playlistSize, 0);
                break;
            case 2: // Shuffle and Play Main Playlist
                interactivePlayback(playlist, playlistSize, 1);
                break;
            case 3: // Show Playlist
                showPlaylist(playlist, playlistSize);
                system("pause");
                break;
            case 4: // Mood-based Recommendation
                moodBasedRecommendation(playlist, playlistSize);
                break;
            case 5: // Genre-based Recommendation
                genreBasedRecommendation(playlist, playlistSize);
                break;
            case 6: // Create User Playlist
                createUserPlaylist(playlist, playlistSize, userPlaylist, &userPlaylistSize);
                break;
            case 7: // Play User Playlist in Queue
                if (userPlaylistSize > 0)
                    interactivePlayback(userPlaylist, userPlaylistSize, 0);
                else {
                    printf("No custom playlist created yet.\n");
                    Sleep(1000);
                }
                break;
            case 8: // Shuffle User Playlist
                if (userPlaylistSize > 0)
                    interactivePlayback(userPlaylist, userPlaylistSize, 1);
                else {
                    printf("No custom playlist created yet.\n");
                    Sleep(1000);
                }
                break;

                break;

                return 0;
            default:
                printf("Invalid option. Try again!\n");
                Sleep(1000);
        }
    }
